test("generation reports shared GPIO assignments across LCD, encoder, and direct sketch controls", () => {
  const project = structuredClone(COMPONENT_EXAMPLES["ky-040"]());
  assert.deepEqual(peripheralPinConflictIssues(project, "uno", project.code), [], "valid encoder pinMode calls must be recognized as setup for the same wired encoder signals");
  project.components.push(
    { id: "lcd", type: "lcd-16x2", label: "Room LCD", x: 600, y: 0 },
    { id: "buzzer", type: "buzzer", label: "Alert buzzer", x: 900, y: 0 },
  );
  for (const [id, lcdPin, boardPin] of [
    ["lcd-rs", "RS", "D12"],
    ["lcd-enable", "E", "D11"],
    ["lcd-d7", "D7", "D2"],
    ["lcd-d6", "D6", "D3"],
    ["lcd-d5", "D5", "D4"],
    ["lcd-d4", "D4", "D5"],
  ]) {
    project.connections.push({ id, from: { componentId: "lcd", pin: lcdPin }, to: { componentId: "uno", pin: boardPin } });
  }
  project.connections.push({ id: "buzzer-positive", from: { componentId: "buzzer", pin: "+" }, to: { componentId: "uno", pin: "D5" } });
  const code = `#include <LiquidCrystal.h>
#include <Encoder.h>
LiquidCrystal lcd(12, 11, 5, 4, 3, 2);
Encoder myEnc(2, 3);
const int buzzerPin = 5;
const int encSwPin = 4;
void setup() { pinMode(buzzerPin, OUTPUT); pinMode(encSwPin, INPUT_PULLUP); }
void loop() { tone(buzzerPin, 880); digitalRead(encSwPin); }`;
  const issues = peripheralPinConflictIssues(project, "uno", code);
  assert.equal(issues.length, 4, issues.join("\n"));
  assert.ok(issues.some(issue => issue.includes("D2") && issue.includes("LCD (lcd-16x2)") && issue.includes("KY-040 (ky-040)") && issue.includes("Room LCD.D7") && issue.includes("wire lcd-d7")));
  assert.ok(issues.some(issue => issue.includes("D4") && issue.includes("encSwPin") && issue.includes("Room LCD.D5")), issues.join("\n"));
  assert.ok(issues.some(issue => issue.includes("D5") && issue.includes("buzzerPin") && issue.includes("Alert buzzer.+")));

  const mismatched = structuredClone(COMPONENT_EXAMPLES["ky-040"]());
  mismatched.code = mismatched.code.replace("Encoder knob(2,3)", "Encoder knob(7,8)");
  const miswireIssues = peripheralPinConflictIssues(mismatched, "uno", mismatched.code);
  assert.equal(miswireIssues.filter(issue => issue.includes("GPIO_PIN_NOT_WIRED")).length, 2, miswireIssues.join("\n"));
});

test("code-only recovery is allowed only after the candidate topology passes electrical validation", () => {
  const sketchIssue = { code: "UNSUPPORTED_CALL", message: "invalid sketch API", componentIds: ["uno"], wireIds: [], nets: [], expectedTopology: "", stage: "code" as const };
  const wiringIssue = { ...sketchIssue, code: "LED_SERIES_RESISTOR_BYPASSED", stage: "wiring" as const };
  assert.equal(canRecoverSketchOnly(false, 1, [sketchIssue]), false, "a malformed sketch stage alone does not prove the assembled wiring is safe");
  assert.equal(canRecoverSketchOnly(true, 1, [sketchIssue]), true, "a validated single-board topology can use source-only recovery");
  assert.equal(canRecoverSketchOnly(true, 1, [sketchIssue, wiringIssue]), false, "any wiring diagnostic requires full-project recovery");
  assert.equal(canRecoverSketchOnly(true, 2, [sketchIssue]), false, "multi-board recovery remains project-wide");
});

test("recovery topology probing detects an LED resistor bypass even before a valid sketch exists", () => {
  const project = greenhouseExample();
  const led = project.components.find(component => component.type === "led")!;
  const candidate = { project, explanation: "Greenhouse", assumptions: [], warnings: [] };
  const context = {
    target: "simulation" as const,
    prompt: greenhousePrompt,
    components: [...new Map(project.components.map(component => [component.type, COMPONENT_CATALOG[component.type]])).values()],
    multipleBoards: false,
  };
  const validProbe = probeRecoveryTopology(candidate, context);
  assert.equal(validProbe.topologyValidated, true, JSON.stringify(validProbe.issues));

  const bypassed = structuredClone(project);
  bypassed.connections.push({ id: "test-led-direct-drive", from: { componentId: "uno", pin: "D6" }, to: { componentId: led.id, pin: "A" } });
  const invalidProbe = probeRecoveryTopology({ ...candidate, project: bypassed }, context);
  assert.equal(invalidProbe.topologyValidated, false);
  assert.ok(invalidProbe.issues.some(issue => issue.includes("LED_SERIES_RESISTOR_BYPASSED")), JSON.stringify(invalidProbe.issues));
  assert.ok(invalidProbe.diagnostics.some(diagnostic => diagnostic.code === "LED_SERIES_RESISTOR_BYPASSED" && diagnostic.stage === "wiring"));

  const collapsedLedTerminals = structuredClone(project);
  collapsedLedTerminals.connections.push({
    id: "greenhouse-led-terminals-shorted",
    from: { componentId: led.id, pin: "A" },
    to: { componentId: led.id, pin: "K" },
  });
  const shortedProbe = probeRecoveryTopology({ ...candidate, project: collapsedLedTerminals }, context);
  assert.equal(shortedProbe.topologyValidated, false, "a malformed staged sketch must not freeze a net with a shorted LED");
  assert.ok(shortedProbe.diagnostics.some(diagnostic => diagnostic.code === "NET_COMPONENT_SHORT" && diagnostic.stage === "wiring"), JSON.stringify(shortedProbe.diagnostics));
});

test("generation separates an unambiguous NeoPixel data resistor from an LCD GPIO conflict", () => {
  const project = structuredClone(COMPONENT_EXAMPLES["ky-040"]());
  project.components.push(
    { id: "lcd", type: "lcd-16x2", label: "Room LCD", x: 400, y: 0 },
    { id: "strip", type: "ws2812b-strip-8", label: "Status strip", x: 800, y: 0 },
    { id: "data-resistor", type: "resistor", label: "WS2812 data resistor", x: 650, y: 0, properties: { resistance: 330 } },
  );
  project.connections.push(
    { id: "lcd-data-collision", from: { componentId: "lcd", pin: "D4" }, to: { componentId: "uno", pin: "D4" } },
    { id: "strip-data", from: { componentId: "strip", pin: "DIN" }, to: { componentId: "data-resistor", pin: "2" } },
    { id: "resistor-source", from: { componentId: "data-resistor", pin: "1" }, to: { componentId: "uno", pin: "D4" } },
  );
  const code = `#include <LiquidCrystal.h>
#include <Adafruit_NeoPixel.h>
LiquidCrystal lcd(2, 3, 4, 5, 6, 7);
Adafruit_NeoPixel strip(8, 4, NEO_GRB + NEO_KHZ800);
void setup(){ lcd.begin(16, 2); strip.begin(); }
void loop(){ strip.show(); }`;
  assert.ok(peripheralPinConflictIssues(project, "uno", code).some(issue => issue.includes("GPIO_PIN_CONFLICT")));

  const repaired = repairConflictedNeoPixelDataBranch(project, "uno", code);
  assert.equal(repaired.repairs.length, 1, repaired.repairs.join("\n"));
  assert.match(repaired.code, /Adafruit_NeoPixel\s+strip\(8,\s*(\d+)/);
  const match = /Adafruit_NeoPixel\s+strip\(8,\s*(\d+)/.exec(repaired.code);
  assert.ok(match);
  const newPin = Number(match[1]);
  assert.notEqual(newPin, 4);
  assert.ok(repaired.connections.some(connection => [connection.from, connection.to].some(endpoint => endpoint.componentId === "data-resistor" && endpoint.pin === "1")
    && [connection.from, connection.to].some(endpoint => endpoint.componentId === "uno" && endpoint.pin === `D${newPin}`)));
  assert.equal(repaired.connections.find(connection => connection.id === "lcd-data-collision")?.to.pin, "D4", "the independent LCD signal stays on its original pin");
  assert.equal(peripheralPinConflictIssues({ ...project, connections: repaired.connections }, "uno", repaired.code).some(issue => issue.includes("GPIO_PIN_CONFLICT")), false);
});

test("a request for an unsupported capacitor gets a clear alternative before model calls", async context => {
  const previousFetch = globalThis.fetch;
  const previousKey = process.env.GEMINI_API_KEY;
  context.after(() => {
    globalThis.fetch = previousFetch;
    if (previousKey === undefined) delete process.env.GEMINI_API_KEY;
    else process.env.GEMINI_API_KEY = previousKey;
  });
  process.env.GEMINI_API_KEY = "test-secret";
  let providerCalls = 0;
  globalThis.fetch = async () => {
    providerCalls += 1;
    return new Response("unexpected model call", { status: 500 });
  };
  const response = await POST(new Request("http://localhost/api/ai/generate", {
    method: "POST",
    body: JSON.stringify({ prompt: "Create a WS2812B strip circuit with its required bulk capacitor." }),
  }));
  const body = await response.json();
  assert.equal(response.status, 422);
  assert.equal(body.error.code, "COMPONENT_UNAVAILABLE");
  assert.match(body.error.message, /capacitor is not in Cirkitra's supported component catalog/);
  assert.ok(body.error.details.some((detail: string) => /add it as a physical hardware note after export/.test(detail)));
  assert.equal(providerCalls, 0, "unsupported parts should be identified before spending time on generation retries");
});

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);

function isFixtureEndpoint(value: unknown): value is FixtureEndpoint {
  return isRecord(value) && typeof value.componentId === "string" && typeof value.pin === "string";
}

function isFixtureWire(value: unknown): value is FixtureWire {
  return isRecord(value) && isFixtureEndpoint(value.from) && isFixtureEndpoint(value.to);
}

function isFixtureEnvelope(value: unknown): value is FixtureEnvelope {
  return isRecord(value) && isRecord(value.project) && Array.isArray(value.project.components) && value.project.components.every(isFixtureComponent);
}

async function responseEnvelope(response: Response): Promise<FixtureEnvelope | undefined> {
  try {
    const payload: unknown = await response.clone().json();
    if (!isRecord(payload) || !Array.isArray(payload.candidates)) return undefined;
    const candidate = payload.candidates[0];
    if (!isRecord(candidate) || !isRecord(candidate.content) || !Array.isArray(candidate.content.parts)) return undefined;
    const text = candidate.content.parts.filter(isRecord).map(part => typeof part.text === "string" ? part.text : "").join("");
    const parsed: unknown = JSON.parse(text);
    return isFixtureEnvelope(parsed) ? parsed : undefined;
  } catch {
    return undefined;
  }
}

function planFromEnvelope(envelope: FixtureEnvelope): FixturePlan | undefined {
  const project = envelope.project;
  const components = project.components;
  const boards: FixtureBoard[] = components.filter(part => isBoardType(part.type)).map(part => ({ id: part.id, type: part.type, label: part.label ?? part.type }));
  if (!boards.length && typeof project.board === "string" && isBoardType(project.board)) boards.push({ id: "board", type: project.board, label: "Board" });
  if (!boards.length) return undefined;
  const nonBoards = components.filter(part => !isBoardType(part.type));
  const subsystemId = "main";
  const parent = new Map<string, string>();
  const find = (key: string): string => {
    const root = parent.get(key) ?? key;
    if (root === key) { parent.set(key, key); return key; }
    const result = find(root); parent.set(key, result); return result;
  };
  const join = (a: string, b: string) => { const ra = find(a); const rb = find(b); if (ra !== rb) parent.set(rb, ra); };
  const connections = (project.connections ?? []).filter(isFixtureWire);
  for (const wire of connections) join(`${wire.from.componentId}:${wire.from.pin}`, `${wire.to.componentId}:${wire.to.pin}`);
  const groups = new Map<string, Array<{ componentId: string; pin: string }>>();
  for (const wire of connections) for (const endpoint of [wire.from, wire.to]) {
    const root = find(`${endpoint.componentId}:${endpoint.pin}`);
    const items = groups.get(root) ?? [];
    if (!items.some(item => item.componentId === endpoint.componentId && item.pin === endpoint.pin)) items.push({ componentId: endpoint.componentId, pin: endpoint.pin });
    groups.set(root, items);
  }
  const nets = [...groups.values()].filter(endpoints => endpoints.length > 1).map((_, index) => ({ id: `net_${index + 1}`, name: `Net ${index + 1}`, subsystemIds: [subsystemId] }));
  return {
    projectId: typeof project.id === "string" && /^[A-Za-z][A-Za-z0-9_-]{0,63}$/.test(project.id) ? project.id : "mock_project",
    name: typeof project.name === "string" ? project.name : "Mock circuit",
    description: typeof project.description === "string" ? project.description : "Generated fixture circuit",
    boards,
    components: nonBoards.map(part => ({ id: part.id, type: part.type, label: part.label ?? part.type, subsystemId })),
    subsystems: [{ id: subsystemId, name: "Main circuit", goal: "Connect and operate the requested components." }],
    nets,
    estimatedNetTerminals: [...groups.values()].filter(endpoints => endpoints.length > 1).reduce((sum, endpoints) => sum + endpoints.length, 0),
    requirements: [{ id: "requested_behavior", description: typeof envelope.explanation === "string" ? envelope.explanation : "Implement the requested behavior." }],
  };
}

function fragmentFromEnvelope(envelope: FixtureEnvelope, plan: FixturePlan, subsystemId: string) {
  const project = envelope.project;
  const projectParts = new Map(project.components.map(part => [part.id, part]));
  const placements = plan.components.filter(part => part.subsystemId === subsystemId).map(part => {
    const source = projectParts.get(part.id);
    if (!source) return undefined;
    const properties = Object.fromEntries(Object.entries(source.properties ?? {}).filter((entry): entry is [string, string | number | boolean] =>
      typeof entry[1] === "string" || typeof entry[1] === "number" || typeof entry[1] === "boolean"));
    return {
      id: part.id,
      x: Math.max(-900, Math.min(900, source.x ?? 0)),
      y: Math.max(-900, Math.min(900, source.y ?? 0)),
      rotation: source.rotation ?? 0,
      properties,
    };
  }).filter((placement): placement is NonNullable<typeof placement> => !!placement);
  const connections = (project.connections ?? []).filter(isFixtureWire);
  const parent = new Map<string, string>();
  const find = (key: string): string => { const root = parent.get(key) ?? key; if (root === key) { parent.set(key, key); return key; } const found = find(root); parent.set(key, found); return found; };
  for (const wire of connections) {
    const left = find(`${wire.from.componentId}:${wire.from.pin}`); const right = find(`${wire.to.componentId}:${wire.to.pin}`); if (left !== right) parent.set(right, left);
  }
  const groups = new Map<string, Array<{ componentId: string; pin: string }>>();
  for (const wire of connections) for (const endpoint of [wire.from, wire.to]) {
    const root = find(`${endpoint.componentId}:${endpoint.pin}`); const entries = groups.get(root) ?? [];
    if (!entries.some(item => item.componentId === endpoint.componentId && item.pin === endpoint.pin)) entries.push({ componentId: endpoint.componentId, pin: endpoint.pin });
    groups.set(root, entries);
  }
  const localNets = [...groups.values()].filter(endpoints => endpoints.length > 1);
  const nets = plan.nets.filter(net => net.subsystemIds.includes(subsystemId)).map((net, index) => ({ id: net.id, endpoints: localNets[index] ?? [] }));
  return { subsystemId, placements, nets };
}

function sketchesFromEnvelope(envelope: FixtureEnvelope, plan: FixturePlan) {
  return structuredFixturePrograms(envelope, plan);
}

const testProviderFetch: typeof fetch = async (input, init) => {
  if (!providerMock || !init?.body) throw new Error("No test provider response was configured");
  const request: unknown = JSON.parse(String(init.body));
  if (!isRecord(request)) throw new Error("Malformed mocked provider request");
  const generationConfig = isRecord(request.generationConfig) ? request.generationConfig : {};
  const responseSchema = isRecord(generationConfig.responseJsonSchema) ? generationConfig.responseJsonSchema : {};
  const schema = isRecord(responseSchema.properties) ? responseSchema.properties : {};
  const contents = Array.isArray(request.contents) ? request.contents : [];
  const firstContent = isRecord(contents[0]) ? contents[0] : {};
  const parts = Array.isArray(firstContent.parts) ? firstContent.parts : [];
  const firstPart = isRecord(parts[0]) ? parts[0] : {};
  const contentText = typeof firstPart.text === "string" ? firstPart.text : "{}";
  let content: Record<string, unknown> = {};
  try { const parsed: unknown = JSON.parse(contentText); if (isRecord(parsed)) content = parsed; } catch { /* Chat requests contain plain text. */ }
  const repairing = Array.isArray(content.validationIssues) && content.validationIssues.length > 0;
  if (schema.projectId) {
    stagedSeed = undefined;
    const response = await providerMock(input, init);
    const envelope = await responseEnvelope(response);
    if (!envelope) return response;
    const plan = planFromEnvelope(envelope);
    if (plan) {
      stagedSeed = { envelope, plan, response };
      return modelResponse(JSON.stringify(plan));
    }
    return response;
  }
  if (schema.project && content.recoveryMode === "whole-project-correction" && stagedSeed) {
    const response = await providerMock(input, init);
    const projectSchema = isRecord(schema.project) ? schema.project : {};
    const projectProperties = isRecord(projectSchema.properties) ? projectSchema.properties : {};
    if (Object.hasOwn(projectProperties, "code")) return response;
    const envelope = await responseEnvelope(response);
    if (envelope) stagedSeed.envelope = envelope;
    return envelope ? modelResponse(JSON.stringify(structuredFixtureResponse(envelope, stagedSeed.plan))) : response;
  }
  if (schema.project && content.recoveryMode === "whole-project" && stagedSeed) {
    const response = await providerMock(input, init);
    const projectSchema = isRecord(schema.project) ? schema.project : {};
    const projectProperties = isRecord(projectSchema.properties) ? projectSchema.properties : {};
    // Sketch-only recovery asks for source code while retaining the already
    // validated topology. Do not reshape that compact response as a new plan.
    if (Object.hasOwn(projectProperties, "code")) return response;
    const envelope = await responseEnvelope(response);
    if (envelope) stagedSeed.envelope = envelope;
    return envelope ? modelResponse(JSON.stringify(structuredFixtureResponse(envelope, stagedSeed.plan))) : response;
  }
  if (schema.project && stagedSeed?.response) {
    const response = stagedSeed.response;
    stagedSeed.response = undefined;
    const envelope = await responseEnvelope(response);
    return envelope ? modelResponse(JSON.stringify(structuredFixtureResponse(envelope, stagedSeed.plan))) : response;
  }
  if (schema.project && stagedSeed) {
    const response = await providerMock(input, init);
    const envelope = await responseEnvelope(response);
    if (envelope) stagedSeed.envelope = envelope;
    return envelope ? modelResponse(JSON.stringify(structuredFixtureResponse(envelope, stagedSeed.plan))) : response;
  }
  if (schema.subsystemId && stagedSeed) {
    const subsystem = isRecord(content.subsystem) ? content.subsystem : {};
    const subsystemId = typeof subsystem.id === "string" ? subsystem.id : stagedSeed.plan.subsystems[0].id;
    if (repairing) {
      const response = await providerMock(input, init);
      const candidate = await responseEnvelope(response);
      if (candidate) stagedSeed.envelope = candidate;
      const fragment = fragmentFromEnvelope(candidate ?? stagedSeed.envelope, stagedSeed.plan, subsystemId);
      if (fragment) return modelResponse(JSON.stringify(fragment));
      return response;
    }
    return modelResponse(JSON.stringify(fragmentFromEnvelope(stagedSeed.envelope, stagedSeed.plan, subsystemId)));
  }
  if (schema.programs && stagedSeed) {
    if (repairing) {
      const response = await providerMock(input, init);
      const candidate = await responseEnvelope(response);
      if (candidate) stagedSeed.envelope = candidate;
      const bundle = sketchesFromEnvelope(candidate ?? stagedSeed.envelope, stagedSeed.plan);

    }

  }
  return providerMock(input, init);
};

    if (Array.isArray(content.validationIssues)) repairRequests += 1;
    return modelResponse(JSON.stringify({ project: changedProject, explanation: "ESP32 reports the received byte and responds to the Mega.", assumptions: [], warnings: [] }));
  };

  const response = await POST(new Request("http://localhost/api/ai/generate", {
    method: "POST",
    headers: { "content-type": "application/json" },
    body: JSON.stringify({
      prompt: "Edit the current Arduino Mega 2560 and ESP32 circuit. Preserve every wire and do not alter wiring. Do not add physical parts. Keep the UART P/A handshake, log each received byte to the ESP32 USB Serial Monitor, and turn on the ESP32 built-in LED on GPIO2 when P arrives.",
      currentProject,
    }),
  }));
  const body = await response.json();
  assert.equal(response.status, 200, JSON.stringify(body));
  assert.equal(repairRequests, 0, "the app should deterministically preserve explicitly protected wiring instead of spending another provider call");
  assert.deepEqual(body.project.connections.map((wire: FixtureWire) => [wire.from.componentId, wire.from.pin, wire.to.componentId, wire.to.pin]), [
    ["mega", "D18", "esp", "GPIO16"],
    ["esp", "GPIO17", "mega", "D19"],
    ["mega", "GND", "esp", "GND"],
  ]);
  assert.equal(body.project.components.length, 2, "edit keeps the same two boards");

  const simulator = new MultiBoardSimulator();
  simulator.attachProject(body.project);
  simulator.run();
  for (let elapsed = 0; elapsed <= 2500; elapsed += 16.67) simulator.advance(16.67);
  const snapshot = simulator.getSnapshot();
  assert.ok(snapshot.boardSerial?.esp?.some(entry => entry.text === "80"), "ESP32 should receive the P byte through the preserved D18/GPIO16 wire");
  assert.ok(snapshot.boardSerial?.mega?.some(entry => entry.text === "65"), `Mega should receive the A acknowledgement through the preserved GPIO17/D19 wire: ${JSON.stringify({ serial: snapshot.boardSerial, diagnostics: snapshot.diagnostics, programs: body.project.programs })}`);
});
